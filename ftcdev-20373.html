// BioBuzz Codebase Concepts

How this TeamCode repo is organized

Start with the calls you make in a program: create profile-managed mechanisms, update driver input, run a shared base, schedule paths, and validate safely. Each concept card opens caller-side examples before its supporting implementation notes.

📚
Library Layer

Call shared helpers such as Timer, GamepadEx, DriveVector, filters, and wrappers from an OpMode instead of rewriting utility behavior locally.

🧩
Constants & Robot Profiles

Start an OpMode with Constants.createRobot(hardwareMap), then use typed require(...) or optional get(...) calls.

🧠
Blackboard & Pose Handoff

Call Blackboard.put(...) when a lifecycle flow owns shared state, then read it in a concrete variant with an explicit default-pose fallback.

🏗️
OpMode Structure

Call a shared base by extending it in a small annotated variant; do not copy whole shared loops into each Driver Station entry.

🟣
Kotlin in TeamCode

Kotlin calls the same Constants.createFollower(hardwareMap) and follower APIs as Java; it does not use a separate robot architecture.

🧪
Tests, Tuners & Ivy

Start with a no-motion factory test or a focused test OpMode, then call planner/Ivy/tuning flows only after the basic profile-created hardware path works.

Real Code, Ready to Use

Copy the call patterns that belong inside an OpMode. Each tab shows how to use an existing TeamCode surface; the detailed chapters explain the implementation only when it helps. 

Robot robot = Constants.createRobot(hardwareMap);
RobotDrive drive = robot.require(RobotDrive.class);
Intake intake = robot.get(Intake.class); // Optional by profile.
Limelight limelight = robot.get(Limelight.class);

telemetry.addData("Registered", robot.getConfiguredMechanismClasses());
telemetry.addData("Has intake", intake != null);
telemetry.update();

// For a follower-only operation:
Follower follower = Constants.createFollower(hardwareMap);
follower.setStartingPose(Constants.RED_DEFAULT_START_POSE);

// A configuration or autonomous flow writes state deliberately.
Blackboard.put(Blackboard.Caption.ALLIANCE, AllianceColor.RED);
Blackboard.put(Blackboard.Caption.POSE, follower.getPose().copy());

// A variant reads the stored pose, then falls back when it is still zero.
Pose saved = Blackboard.get(Blackboard.Caption.POSE).getValue();
if (saved.getX() != 0 || saved.getY() != 0 || saved.getHeading() != 0) {
    drive.setStartingPose(saved);
} else {
    drive.setStartingPose(Constants.RED_DEFAULT_START_POSE);
}
}
GamepadEx driver = new GamepadEx(gamepad1);
GpBindings bindings = driver.result();
Robot robot = Constants.createRobot(hardwareMap);
RobotDrive drive = robot.require(RobotDrive.class);

drive.setStartingPose(Constants.RED_DEFAULT_START_POSE);
drive.setStickCurve(x -> Math.signum(x) * x * x);
drive.startTeleOpDrive();
waitForStart();

while (opModeIsActive()) {
    driver.update();
    robot.periodic(bindings);
    telemetry.addData("Pose", drive.getPose());
    telemetry.update();
}
}
@TeleOp(name = "Kotlin TeleOp", group = "Kotlin")
class KotlinTeleOpTest: LinearOpMode() {
    lateinit var follower: Follower
    override fun runOpMode() {
        follower = Constants.createFollower(hardwareMap)
        follower.setStartingPose(Pose())
        waitForStart()
        follower.startTeleOpDrive()
        while (opModeIsActive()) {
            follower.setTeleOpDrive(-gamepad1.left_stick_y.toDouble(), -gamepad1.left_stick_x.toDouble(), -gamepad1.right_stick_x.toDouble(), false)
            follower.update()
            if (gamepad1.back) {
                follower.heading = 0.0;
            }

            telemetry.addLine("FIELD CENTRIC KOTLIN TELEOP")
            telemetry.addData("Yaw", Math.toDegrees(follower.heading))
            telemetry.update()
        }
    }
}
@Autonomous(name = "My Red Auto")
public class MyRedAuto extends IvyAuton {
    static class Paths extends IvyAuton_Paths {
        Pose start = new Pose(36, 12, Math.toRadians(180));
        @Override protected Pose getStartingPose() { return start.copy(); }
        @Override protected PathChain getPath1() {
            Path line = new Path(new BezierLine(start, new Pose(36, 36)));
            line.setHeadingInterpolation(
                HeadingInterpolator.linear(start.getHeading(), Math.toRadians(180)));
            return new PathChain(line);
        }
    }
    public MyRedAuto() { super(AllianceColor.RED, new Paths()); }
}
@Override
protected void test(HardwareMap hardwareMap, GamepadEx gamepad1,
        GamepadEx gamepad2, Telemetry telemetry) {
    Follower follower = Constants.createFollower(hardwareMap);
    follower.setStartingPose(Constants.RED_DEFAULT_START_POSE);
    follower.update();
    follower.startTeleopDrive();
    follower.setTeleOpDrive(0, 0, 0, false);

    telemetry.addData("Pose", follower.getPose());
    telemetry.addData("Velocity", follower.getVelocity());
    telemetry.addLine("This construction test should not move the robot.");
    telemetry.update();
    }
}
